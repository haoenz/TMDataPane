cmake_minimum_required(VERSION 3.24)
project(DataPane LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "DataPane currently targets Windows x64 with MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/TrafficMonitor/PluginInterface.h")
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 first")
endif()
add_library(DataPane SHARED src/Plugin.Native/Plugin.cpp)
target_include_directories(DataPane SYSTEM PRIVATE third_party/TrafficMonitor)
target_compile_definitions(DataPane PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(DataPane PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(DataPane PRIVATE ole32)
add_executable(LifecycleHost tests/LifecycleHost/main.cpp)
target_include_directories(LifecycleHost SYSTEM PRIVATE third_party/TrafficMonitor)
target_compile_definitions(LifecycleHost PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LifecycleHost PRIVATE /W4 /utf-8 /EHsc)
